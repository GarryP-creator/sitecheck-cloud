// ============================================================================
// SiteCheck — admin-users
//
// Creating a login needs the service role key, which can do anything and must
// never be sent to a browser. So it lives here instead, on Supabase, and the
// app asks this function to act on its behalf.
//
// Every request is checked twice: the caller must be signed in, and their
// profile must say superuser. Neither check can be skipped from the app side,
// because both happen here.
//
// Deploy: Supabase dashboard -> Edge Functions -> Deploy a new function,
// name it exactly "admin-users", paste this in, deploy.
// ============================================================================

import { createClient } from 'jsr:@supabase/supabase-js@2';

const DOMAIN = 'wrm.internal';

const cors = {
  'Access-Control-Allow-Origin': '*',
  'Access-Control-Allow-Headers': 'authorization, x-client-info, apikey, content-type',
  'Access-Control-Allow-Methods': 'POST, OPTIONS',
};

const json = (body: unknown, status = 200) =>
  new Response(JSON.stringify(body), {
    status,
    headers: { ...cors, 'Content-Type': 'application/json' },
  });

const validId = (id: string) => /^[A-Za-z0-9._-]{3,32}$/.test(id ?? '');
const emailFor = (id: string) => `${id.trim().toLowerCase()}@${DOMAIN}`;

// --- what makes an acceptable password --------------------------------------
// The same rules as pwProblem in cloud.js, and this is the copy that matters:
// the browser's is there to tell somebody what is wrong as they type, this one
// is there because the browser's can be skipped. Change one, change the other.
//
// Length first, composition rules not at all. Insisting on a capital, a digit
// and a symbol produces Passw0rd! and a note on the hoarding.
const BANNED = [
  'password', 'passw0rd', 'pa55word', 'sitecheck', 'wrm', 'workplace', 'letmein',
  'welcome', 'qwerty', 'qwertyuiop', 'asdfgh', 'zxcvbn', 'iloveyou', 'monkey',
  'dragon', 'football', 'liverpool', 'manchester', 'arsenal', 'chelsea', 'rangers',
  'celtic', 'admin', 'changeme', 'trustno1', 'sunshine', 'princess', 'superman',
  'scaffold', 'hardhat', 'builder', 'construction',
];

// Four or more characters running in order, up or down: 1234, 4321, abcd.
const hasRun = (s: string, len: number) => {
  let up = 1, down = 1;
  for (let i = 1; i < s.length; i++) {
    const d = s.charCodeAt(i) - s.charCodeAt(i - 1);
    up = d === 1 ? up + 1 : 1;
    down = d === -1 ? down + 1 : 1;
    if (up >= len || down >= len) return true;
  }
  return false;
};

// Returns a sentence to show the person, or null if the password is fine.
const pwProblem = (pw: unknown, userId: string): string | null => {
  const p = typeof pw === 'string' ? pw : '';
  if (!p) return 'Enter a password';
  if (p !== p.trim()) return 'The password cannot start or end with a space';
  if (p.length < 10) return 'The password must be at least 10 characters';
  if (p.length > 72) return 'That password is too long — 72 characters at most';
  if (/^\d+$/.test(p)) return 'Not all numbers — that is only a longer PIN';

  const low = p.toLowerCase();
  const id = (userId ?? '').trim().toLowerCase();
  if (id.length >= 3 && low.includes(id)) return 'The password cannot contain their User ID';

  const hit = BANNED.find((b) => low.includes(b));
  if (hit) return `"${hit}" is one of the first things anyone would try`;

  // Counting variety rather than banning a repeat: "allsmalllettersok" has lll
  // in it and is a perfectly good password. What these catch is aaaaaaaaaa.
  if (new Set(low).size < 5) return 'Too few different characters in that password';
  if (/(.)\1{3,}/.test(p)) return 'Do not repeat one character four times over';
  if (hasRun(low, 4)) return 'Do not use a run like 1234 or abcd';
  return null;
};

Deno.serve(async (req) => {
  if (req.method === 'OPTIONS') return new Response('ok', { headers: cors });

  // Supabase has changed the name of the privileged key more than once, so
  // take whichever is present rather than assuming. Without it, this function
  // can do nothing useful — better to say so than to fail as "not allowed".
  // SITECHECK_SERVICE_KEY comes first deliberately. On projects using the new
  // sb_secret_ key format, the built-in SUPABASE_SERVICE_ROLE_KEY still exists
  // but is the old JWT and no longer carries any privileges — so it is found,
  // used, and quietly refused. An explicit secret avoids that trap entirely.
  // Which key we ended up with matters, and guessing at it wastes time — so
  // record the source and report it when something is refused.
  const KEY_SOURCES = ['SITECHECK_SERVICE_KEY', 'SUPABASE_SECRET_KEY', 'SUPABASE_SERVICE_ROLE_KEY'];
  const keyFrom = KEY_SOURCES.find((n) => !!Deno.env.get(n));
  const SERVICE_KEY = keyFrom ? Deno.env.get(keyFrom) : undefined;

  // A fingerprint, not the key: enough to tell which one is in use, useless
  // to anyone who sees it.
  const keyShape = SERVICE_KEY
    ? `${SERVICE_KEY.slice(0, 10)}…${SERVICE_KEY.length} chars`
    : 'none';

  if (!SERVICE_KEY) {
    return json({
      error: 'No service key. Add a secret named SITECHECK_SERVICE_KEY in Edge Functions -> Secrets, using the sb_secret_ key from Settings -> API Keys.',
    }, 500);
  }

  const URL = Deno.env.get('SUPABASE_URL')!;

  // Two separate clients, deliberately. Verifying a user's token on the same
  // client that queries the database can leave that client using the user's
  // token afterwards — which then gets refused, and looks like a key problem
  // when it is not. Keeping them apart removes the ambiguity.
  const admin = createClient(URL, SERVICE_KEY, {
    auth: { persistSession: false, autoRefreshToken: false },
    global: { headers: { apikey: SERVICE_KEY, Authorization: `Bearer ${SERVICE_KEY}` } },
  });

  // Database reads and writes go through plain REST with the service key on
  // every request. Nothing can quietly swap the credentials underneath us.
  const db = async (path: string, init: RequestInit = {}) => {
    const res = await fetch(`${URL}/rest/v1/${path}`, {
      ...init,
      headers: {
        apikey: SERVICE_KEY,
        Authorization: `Bearer ${SERVICE_KEY}`,
        'Content-Type': 'application/json',
        Prefer: 'return=representation',
        ...(init.headers ?? {}),
      },
    });
    const text = await res.text();
    let body: unknown = null;
    try { body = text ? JSON.parse(text) : null; } catch { body = text; }
    return { ok: res.ok, status: res.status, body };
  };

  // --- the wording ---------------------------------------------------------
  // Kept in the database so a superuser can change it from inside the app.
  // These defaults are the safety net: if the row is missing or empty, an
  // email still goes out with sensible words rather than nothing at all.
  // They are duplicated in index.html behind "Reset to the default wording" —
  // change one, change the other.
  const DEFAULT_SUBJECT = 'Your SiteCheck login';
  const DEFAULT_FROM = 'SiteCheck <sitecheck@workplace-rm.com>';
  const DEFAULT_LINK = 'https://garryp-creator.github.io/sitecheck-cloud/';
  const DEFAULT_BODY = [
    'Hello [name],',
    '',
    "You have been set up with an account on SiteCheck, WRM's site compliance app.",
    '',
    'It holds the site checklists, toolbox talks, risk assessments and incident reports, and lets you sign them off on site.',
    '',
    'Open it here: [link]',
    '',
    '*Your details*',
    'User ID: *[userid]*',
    'Temporary password: *[password]*',
    '',
    'Enter your User ID and that password. It works once: the app will then ask you to choose a password of your own, at least 10 characters long. Nobody at WRM can see what you choose.',
    '',
    'Five wrong attempts will lock you out for a few minutes.',
    '',
    '*Put it on your phone*',
    'On iPhone, open the link in Safari, tap Share, then Add to Home Screen.',
    'On Android, open the link in Chrome, tap the menu, then Install app.',
    'It then sits alongside your other apps and opens straight to the sign-in screen.',
    '',
    'Keep your password to yourself. Anything you sign carries your name and cannot be altered afterwards.',
    '',
    'Any problems, speak to Garry Pougher (garry@workplace-rm.com).',
    '',
    'Kind regards,',
    '*WRM* · Workplace Risk Management Ltd',
    'SiteCheck — the whole site, in your pocket',
  ].join('\n');

  const readSettings = async () => {
    const res = await db('app_settings?select=key,value');
    const got: Record<string, string> = {};
    if (res.ok && Array.isArray(res.body)) {
      for (const r of res.body as Array<{ key: string; value: string | null }>) {
        got[r.key] = r.value ?? '';
      }
    }
    return {
      welcome_subject: got.welcome_subject || DEFAULT_SUBJECT,
      welcome_body: got.welcome_body || DEFAULT_BODY,
      welcome_from: got.welcome_from || DEFAULT_FROM,
      welcome_link: got.welcome_link || DEFAULT_LINK,
    };
  };

  // --- turning the wording into an email ------------------------------------
  // The text is written by a person in a text box, so it is treated as text:
  // escaped first, then given the small amount of shape it is allowed to have.
  // Blank line starts a paragraph, *stars* make bold, a bare address becomes a
  // link. Nothing else. This is what keeps a typo from becoming broken HTML.
  // [pin] is still understood and filled with the password. Wording saved
  // before passwords arrived would otherwise come out with an empty gap where
  // the login used to be, which is the one failure nobody would notice until
  // somebody could not get in.
  const fill = (s: string, v: Record<string, string>) =>
    s.replace(/\[(name|userid|password|pin|link)\]/g,
      (_m, k) => v[k === 'pin' ? 'password' : k] ?? '');

  const escHtml = (s: string) =>
    s.replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;');

  const C = { ink: '#14171A', hivis: '#FFD400', plate: '#FBFAF7', line: '#C9C6BE', grey: '#6B6B66' };

  const bodyHtml = (text: string) =>
    text
      .split(/\n\s*\n/)
      .map((para) => {
        const inner = escHtml(para.trim())
          .replace(/(https?:\/\/[^\s<]+)/g,
            `<a href="$1" style="color:${C.ink};font-weight:600">$1</a>`)
          .replace(/\*([^*\n]+)\*/g, '<strong>$1</strong>')
          .replace(/\n/g, '<br>');
        return `<p style="margin:0 0 16px;font-size:15px;line-height:1.55;color:${C.ink}">${inner}</p>`;
      })
      .join('');

  // Deliberately plain. Email clients disagree about almost everything, and a
  // striped masthead built from CSS gradients is one of the things they
  // disagree about — so the hazard tape here is a single solid hi-vis rule,
  // which every client renders identically.
  const wrapHtml = (text: string) => `
<div style="background:${C.plate};padding:24px 12px;font-family:-apple-system,BlinkMacSystemFont,'Segoe UI',Roboto,Helvetica,Arial,sans-serif">
  <div style="max-width:560px;margin:0 auto;background:#FFFFFF;border:1px solid ${C.line}">
    <div style="background:${C.ink};padding:18px 24px">
      <span style="color:${C.hivis};font-size:20px;font-weight:700;letter-spacing:0.5px">SiteCheck</span>
    </div>
    <div style="height:6px;background:${C.hivis}"></div>
    <div style="padding:28px 24px 8px">${bodyHtml(text)}</div>
    <div style="border-top:1px solid ${C.line};padding:14px 24px;font-size:11px;color:${C.grey}">
      Sent by SiteCheck &middot; WRM &middot; Workplace Risk Management Ltd
    </div>
  </div>
</div>`;

  // The plain-text copy that rides alongside. Stars are for emphasis on screen,
  // not for reading, so they come out.
  const bodyText = (text: string) => text.replace(/\*([^*\n]+)\*/g, '$1');

  const RESEND_KEY = Deno.env.get('SITECHECK_RESEND_KEY');

  const sendMail = async (to: string, subject: string, text: string, from: string) => {
    if (!RESEND_KEY) {
      return { ok: false, error: 'No email key. Add a secret named SITECHECK_RESEND_KEY in Edge Functions -> Secrets, using the key from resend.com.' };
    }
    if (!to || !/^[^@\s]+@[^@\s]+\.[^@\s]+$/.test(to)) {
      return { ok: false, error: 'No usable email address for that person' };
    }
    let res: Response;
    try {
      res = await fetch('https://api.resend.com/emails', {
        method: 'POST',
        headers: { Authorization: `Bearer ${RESEND_KEY}`, 'Content-Type': 'application/json' },
        body: JSON.stringify({ from, to: [to], subject, text: bodyText(text), html: wrapHtml(text) }),
      });
    } catch (e) {
      return { ok: false, error: 'Could not reach the email service: ' + (e as Error).message };
    }
    const raw = await res.text();
    let out: any = null;
    try { out = raw ? JSON.parse(raw) : null; } catch { out = raw; }
    if (!res.ok) {
      // Say what the email service actually said. Reporting this as a generic
      // failure is how you lose an afternoon to a domain that was never
      // verified.
      return { ok: false, error: (out && out.message) || `The email service refused it (${res.status})`, detail: out };
    }
    return { ok: true, id: (out && out.id) || null };
  };

  // --- who is asking? ------------------------------------------------------
  const token = (req.headers.get('Authorization') ?? '').replace('Bearer ', '');
  if (!token) return json({ error: 'Not signed in' }, 401);

  const { data: userData, error: userErr } = await admin.auth.getUser(token);
  if (userErr || !userData?.user) {
    return json({ error: 'Could not identify you: ' + (userErr?.message ?? 'no user') }, 401);
  }

  const lookup = await db(`profiles?id=eq.${userData.user.id}&select=role,active,email,full_name,user_id`);

  if (!lookup.ok) {
    return json({
      error: 'Could not read profiles with the service key',
      status: lookup.status,
      detail: lookup.body,
      keySource: keyFrom ?? 'none found',
      keyShape,
      secretsPresent: KEY_SOURCES.filter((n) => !!Deno.env.get(n)),
    }, 500);
  }

  const caller = Array.isArray(lookup.body) ? lookup.body[0] : null;
  if (!caller) {
    return json({ error: 'No profile found for your account (' + userData.user.id + ')' }, 403);
  }
  if (caller.role !== 'superuser' || !caller.active) {
    return json({ error: `Only a superuser can manage accounts — your role is "${caller.role}"` }, 403);
  }

  // --- what are they asking for? -------------------------------------------
  let body: Record<string, any>;
  try { body = await req.json(); }
  catch { return json({ error: 'Bad request' }, 400); }

  const action = body.action ?? 'create';

  // ---------------------------------------------------------------- whoami
  // Says what the function can see. Useful when something is refusing and it
  // is not obvious which of the several gates is shut.
  if (action === 'whoami') {
    return json({
      ok: true,
      you: userData.user.id,
      email: userData.user.email,
      role: caller.role,
      active: caller.active,
      keySource: keyFrom,
      keyShape,
    });
  }

  // ---------------------------------------------------------------- create
  if (action === 'create') {
    const { userId, password, fullName, email, role } = body;

    if (!validId(userId))  return json({ error: 'User ID must be 3+ characters: letters, numbers, dots, hyphens or underscores' }, 400);
    if (!fullName?.trim()) return json({ error: 'Enter their full name' }, 400);
    if (!['site', 'manager', 'superuser'].includes(role ?? '')) return json({ error: 'Unknown role' }, 400);

    const pwBad = pwProblem(password, userId);
    if (pwBad) return json({ error: pwBad }, 400);

    const existing = await db(`profiles?user_id=ilike.${encodeURIComponent(userId)}&select=id`);
    if (existing.ok && Array.isArray(existing.body) && existing.body.length) {
      return json({ error: `${userId} is already in use` }, 400);
    }

    // No pw_set_at in the metadata, deliberately. The app reads its absence as
    // "still on a password somebody else chose" and makes them pick their own
    // before they can reach anything. So the password set here is temporary
    // whether or not the superuser describes it that way.
    const { data: created, error: ce } = await admin.auth.admin.createUser({
      email: emailFor(userId),
      password,
      email_confirm: true,
    });
    if (ce) return json({ error: ce.message }, 400);

    const ins = await db('profiles', {
      method: 'POST',
      body: JSON.stringify({
        id: created.user.id,
        user_id: userId.trim(),
        full_name: fullName.trim(),
        email: email?.trim() || null,
        role,
        created_by: userData.user.id,
      }),
    });

    // if the profile fails, do not leave a login stranded without one
    if (!ins.ok) {
      await admin.auth.admin.deleteUser(created.user.id);
      return json({ error: 'Could not create the profile', detail: ins.body }, 400);
    }

    // The account exists and works from this point on. The email is a
    // courtesy on top, so a failure to send is reported but never undoes the
    // account — being told "could not create" about an account that was in
    // fact created is exactly the kind of wrong answer that wastes hours.
    let mail: Record<string, unknown> = { sent: false };
    if (body.sendWelcome && email?.trim()) {
      const s = await readSettings();
      const text = fill(s.welcome_body, {
        name: (fullName ?? '').trim().split(/\s+/)[0] || fullName || '',
        userid: userId.trim(),
        password,
        link: s.welcome_link,
      });
      const sent = await sendMail(email.trim(), fill(s.welcome_subject, {
        name: fullName, userid: userId, password: '', link: s.welcome_link,
      }), text, s.welcome_from);
      mail = sent.ok ? { sent: true } : { sent: false, error: sent.error, detail: sent.detail };
      if (sent.ok) {
        await db(`profiles?id=eq.${created.user.id}`, {
          method: 'PATCH',
          body: JSON.stringify({ welcome_sent_at: new Date().toISOString() }),
        });
      }
    }

    return json({ ok: true, id: created.user.id, userId, mail });
  }

  // ------------------------------------------------------- reset a password
  // A superuser can give somebody a way back in. What they cannot do is read
  // the password that person was using — only a hash of it exists, here or
  // anywhere else — so "reset" is the only help available, and that is the
  // point rather than a shortcoming.
  //
  // Clearing pw_set_at matters as much as setting the password: without it the
  // person would keep using a password their manager knows. With it, the app
  // stops them at the password screen on the way in.
  if (action === 'setpassword') {
    const { id, userId, password } = body;
    if (!id || !validId(userId)) return json({ error: 'Missing user' }, 400);

    const pwBad = pwProblem(password, userId);
    if (pwBad) return json({ error: pwBad }, 400);

    const { error } = await admin.auth.admin.updateUserById(id, {
      password,
      user_metadata: { pw_set_at: null },
    });
    if (error) return json({ error: error.message }, 400);
    return json({ ok: true });
  }

  // ------------------------------------------------------------- deactivate
  // The login is kept rather than deleted, so their name stays readable on
  // every record they signed. Deactivating simply stops them signing in.
  if (action === 'setactive') {
    const { id, active } = body;
    if (!id) return json({ error: 'Missing user' }, 400);
    if (id === userData.user.id) return json({ error: 'You cannot deactivate yourself' }, 400);

    const upd = await db(`profiles?id=eq.${id}`, {
      method: 'PATCH',
      body: JSON.stringify({ active: !!active }),
    });
    if (!upd.ok) return json({ error: 'Could not update', detail: upd.body }, 400);
    return json({ ok: true });
  }

  // -------------------------------------------------------------- settings
  // What the Welcome email screen loads. emailReady says whether a send would
  // even be attempted, so the screen can say so plainly rather than letting
  // someone write a message that has nowhere to go.
  if (action === 'settings') {
    const s = await readSettings();
    return json({ ok: true, settings: s, emailReady: !!RESEND_KEY, yourEmail: caller.email || null });
  }

  if (action === 'savesettings') {
    const wanted: Record<string, unknown> = {};
    for (const k of ['welcome_subject', 'welcome_body', 'welcome_from', 'welcome_link']) {
      if (typeof body[k] === 'string') wanted[k] = body[k];
    }
    if (!Object.keys(wanted).length) return json({ error: 'Nothing to save' }, 400);
    if (wanted.welcome_subject !== undefined && !String(wanted.welcome_subject).trim()) {
      return json({ error: 'The subject cannot be empty' }, 400);
    }
    if (wanted.welcome_body !== undefined && !String(wanted.welcome_body).trim()) {
      return json({ error: 'The message cannot be empty' }, 400);
    }

    for (const [key, value] of Object.entries(wanted)) {
      const up = await db('app_settings?on_conflict=key', {
        method: 'POST',
        headers: { Prefer: 'resolution=merge-duplicates,return=representation' },
        body: JSON.stringify({ key, value, updated_at: new Date().toISOString(), updated_by: userData.user.id }),
      });
      if (!up.ok) return json({ error: 'Could not save the wording', detail: up.body }, 400);
    }
    return json({ ok: true });
  }

  // -------------------------------------------------------------- test send
  // Sends whatever is currently in the box, unsaved, to the person editing it,
  // with obviously fake details in place of a real login.
  if (action === 'testmail') {
    const s = await readSettings();
    const to = (body.to || caller.email || '').trim();
    if (!to) return json({ error: 'You have no email address on your own profile. Add one first, then try again.' }, 400);
    const text = fill(String(body.welcome_body ?? s.welcome_body), {
      name: (caller.full_name || 'Teresa').split(/\s+/)[0],
      userid: 'SampleT',
      password: 'Example-Not-Real-00',
      link: String(body.welcome_link ?? s.welcome_link),
    });
    const subject = '[Test] ' + String(body.welcome_subject ?? s.welcome_subject);
    const sent = await sendMail(to, subject, text, String(body.welcome_from ?? s.welcome_from));
    if (!sent.ok) return json({ error: sent.error, detail: sent.detail }, 400);
    return json({ ok: true, to });
  }

  // ---------------------------------------------------------------- welcome
  // Sending it again to somebody who already has an account. Their password is
  // not stored anywhere in a form anyone can read — only a hash of it — so
  // there is nothing to remind them of. A resend therefore sets a new
  // temporary password and sends that. The screen says so before it happens.
  if (action === 'welcome') {
    const { id, userId, password } = body;
    if (!id || !validId(userId)) return json({ error: 'Missing user' }, 400);

    const pwBad = pwProblem(password, userId);
    if (pwBad) return json({ error: pwBad }, 400);

    const who = await db(`profiles?id=eq.${id}&select=full_name,email,user_id`);
    const p = who.ok && Array.isArray(who.body) ? who.body[0] : null;
    if (!p) return json({ error: 'No profile found for that person' }, 400);
    if (!p.email) return json({ error: `${p.user_id} has no email address on their profile. Add one first.` }, 400);

    const { error: pe } = await admin.auth.admin.updateUserById(id, {
      password,
      user_metadata: { pw_set_at: null },
    });
    if (pe) return json({ error: 'Could not set the new password: ' + pe.message }, 400);

    const s = await readSettings();
    const text = fill(s.welcome_body, {
      name: (p.full_name || '').trim().split(/\s+/)[0] || p.user_id,
      userid: p.user_id,
      password,
      link: s.welcome_link,
    });
    const sent = await sendMail(p.email, s.welcome_subject, text, s.welcome_from);
    if (!sent.ok) {
      // The password has already changed by this point. Saying so plainly is
      // the difference between a puzzled phone call and a locked-out engineer.
      return json({ error: sent.error + ' — note their password has already been changed to the one you just entered, so tell them another way.', detail: sent.detail }, 400);
    }

    await db(`profiles?id=eq.${id}`, {
      method: 'PATCH',
      body: JSON.stringify({ welcome_sent_at: new Date().toISOString() }),
    });
    return json({ ok: true, to: p.email });
  }

  return json({ error: 'Unknown action' }, 400);
});
